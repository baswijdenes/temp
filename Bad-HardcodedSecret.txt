<#
.SYNOPSIS
    PILLAR 3 -- GOOD example: pull the secret from Azure Key Vault at run
    time, never store it in code.

.DESCRIPTION
    The script never sees the secret value until it asks Key Vault for it,
    and the value lives only in process memory as a SecureString. Nothing
    ends up on disk, in OneDrive, or in your shell history.

    In an Azure-hosted runtime (Function App, Automation Account, VM with
    system-assigned identity, GitHub Actions with OIDC, ...), use:

        Connect-AzAccount -Identity

    so the script holds NO credentials at all -- the platform proves who
    it is for you.

    Docs:
      - Get-AzKeyVaultSecret:
        https://learn.microsoft.com/powershell/module/az.keyvault/get-azkeyvaultsecret
      - Managed identities for Azure resources:
        https://learn.microsoft.com/entra/identity/managed-identities-azure-resources/overview
#>
[CmdletBinding()]
param(
    [Parameter(Mandatory)]
    [ValidateNotNullOrEmpty()]
    [string]$VaultName,

    [ValidateNotNullOrEmpty()]
    [string]$SecretName = 'demo-api-key'
)

Set-StrictMode -Version Latest
$ErrorActionPreference = 'Stop'

# In Azure: Connect-AzAccount -Identity
# Locally for this demo: Connect-AzAccount (already done by setup script)

try {
    $secret = Get-AzKeyVaultSecret -VaultName $VaultName -Name $SecretName -AsPlainText
} catch {
    throw "Could not retrieve secret '$SecretName' from vault '$VaultName': $_"
}

# Use the secret -- but NEVER print it. (We mask the demo output instead.)
$masked = ('*' * ($secret.Length - 4)) + $secret.Substring($secret.Length - 4)
Write-Host "Got secret from $VaultName : $masked" -ForegroundColor Green

# Invoke-RestMethod -Uri 'https://api.example.com/...' -Headers @{ 'x-api-key' = $secret }

<#
    Demo invocations -- swap kv-defps-demo-xxxxxx for the real vault
    name printed by scripts/setup/New-DemoEnvironment.ps1.

    # 1. Mandatory parameter: no -VaultName -> bind-time failure,
    #    BEFORE any Azure call is made.
    .\Good-KeyVaultSecret.ps1

    # 2. ValidateNotNullOrEmpty rejects empty input.
    .\Good-KeyVaultSecret.ps1 -VaultName ''

    # 3. Wrong vault name -> try/catch wraps it in a clear, actionable
    #    error instead of a noisy stack trace.
    .\Good-KeyVaultSecret.ps1 -VaultName 'kv-does-not-exist'

    # 4. Wrong secret name in a real vault -> same: clean throw.
    .\Good-KeyVaultSecret.ps1 -VaultName 'kv-defps-demo-xxxxxx' -SecretName 'no-such-secret'

    # 5. Real run -- prints only the masked value, not the secret.
    .\Good-KeyVaultSecret.ps1 -VaultName 'kv-defps-demo-xxxxxx'

    # 6. Prove nothing leaked to disk: the file itself contains zero
    #    secret material.
    Select-String -Path .\Good-KeyVaultSecret.ps1 -Pattern 'sk-live'   # 0 matches
#>
